Shannon

shannon <- bp |> 
  select(site_id, taxon, count, total_count) |>
  mutate(pi = count / total_count,
         ln_pi = log(pi),
         pi_ln_pi = pi*ln_pi) |>
  group_by(site_id) |>
  summarize(H = -sum(pi_ln_pi))
